```python
# brain_trader/data/vendors/base.py
from __future__ import annotations

import functools
import logging
import time
from abc import ABC, abstractmethod
from typing import Any, Callable, TypeVar

logger = logging.getLogger(__name__)

_F = TypeVar("_F", bound=Callable[..., Any])


def with_retry(attempts: int = 3, base_delay: float = 2.0) -> Callable[[_F], _F]:
    """Exponential-backoff retry decorator for vendor network calls."""
    def decorator(fn: _F) -> _F:
        @functools.wraps(fn)
        def wrapper(*args: Any, **kwargs: Any) -> Any:
            last_exc: Exception | None = None
            for attempt in range(attempts):
                try:
                    return fn(*args, **kwargs)
                except Exception as exc:
                    last_exc = exc
                    if attempt < attempts - 1:
                        delay = base_delay * (2 ** attempt)
                        logger.warning(
                            "Attempt %d/%d failed (%s). Retrying in %.1fs.",
                            attempt + 1, attempts, exc, delay,
                        )
                        time.sleep(delay)
            raise last_exc  # type: ignore[misc]
        return wrapper  # type: ignore[return-value]
    return decorator


class BaseDataVendor(ABC):
    """Interface every data vendor must implement.

    All methods return a plain string formatted for LLM consumption.
    On data unavailability, return a structured error string rather than raising
    so analyst agents can note the gap in their report.
    """

    @abstractmethod
    def get_stock_data(self, ticker: str, start: str, end: str) -> str: ...

    @abstractmethod
    def get_indicators(
        self, ticker: str, indicators: list[str], start: str, end: str
    ) -> str: ...

    @abstractmethod
    def get_fundamentals(self, ticker: str) -> str: ...

    @abstractmethod
    def get_balance_sheet(self, ticker: str) -> str: ...

    @abstractmethod
    def get_cashflow(self, ticker: str) -> str: ...

    @abstractmethod
    def get_income_statement(self, ticker: str) -> str: ...

    @abstractmethod
    def get_news(self, ticker: str, date: str) -> str: ...

    @abstractmethod
    def get_global_news(self, date: str) -> str: ...

    @abstractmethod
    def get_insider_transactions(self, ticker: str) -> str: ...
```

```python
# brain_trader/data/registry.py
from __future__ import annotations

import logging
from brain_trader.config.schema import DataVendorConfig
from brain_trader.data.vendors.base import BaseDataVendor

logger = logging.getLogger(__name__)

# Maps each tool name to its config category field on DataVendorConfig
_TOOL_TO_CATEGORY: dict[str, str] = {
    "get_stock_data": "core_stock",
    "get_indicators": "technical_indicators",
    "get_fundamentals": "fundamental_data",
    "get_balance_sheet": "fundamental_data",
    "get_cashflow": "fundamental_data",
    "get_income_statement": "fundamental_data",
    "get_news": "news_data",
    "get_global_news": "news_data",
    "get_insider_transactions": "news_data",
}


class DataVendorRegistry:
    """Resolves the correct BaseDataVendor implementation for each tool name.

    Vendors are instantiated lazily and cached by vendor name so the same
    yfinance session is reused across all tools that share it.
    """

    def __init__(self, config: DataVendorConfig) -> None:
        self._config = config
        self._cache: dict[str, BaseDataVendor] = {}

    def get_vendor(self, tool_name: str) -> BaseDataVendor:
        vendor_name = self._resolve_vendor_name(tool_name)
        if vendor_name not in self._cache:
            self._cache[vendor_name] = self._instantiate(vendor_name)
        return self._cache[vendor_name]

    def _resolve_vendor_name(self, tool_name: str) -> str:
        if tool_name in self._config.tool_overrides:
            return self._config.tool_overrides[tool_name]
        category = _TOOL_TO_CATEGORY.get(tool_name)
        if category is None:
            raise ValueError(f"Unknown tool: {tool_name!r}")
        return getattr(self._config, category)

    def _instantiate(self, vendor_name: str) -> BaseDataVendor:
        if vendor_name == "yfinance":
            from brain_trader.data.vendors.yfinance import YFinanceVendor
            return YFinanceVendor()
        if vendor_name == "alpha_vantage":
            from brain_trader.data.vendors.alpha_vantage import AlphaVantageVendor
            return AlphaVantageVendor()
        raise ValueError(f"Unknown vendor: {vendor_name!r}")
```

```python
# brain_trader/data/vendors/yfinance.py
from __future__ import annotations

import logging
from datetime import datetime, timedelta

import yfinance as yf

from brain_trader.data.vendors.base import BaseDataVendor, with_retry

logger = logging.getLogger(__name__)

_ERROR = "[DATA UNAVAILABLE: {reason}]"


class YFinanceVendor(BaseDataVendor):

    @with_retry(attempts=3, base_delay=2.0)
    def get_stock_data(self, ticker: str, start: str, end: str) -> str:
        df = yf.download(ticker, start=start, end=end, progress=False, auto_adjust=True)
        if df.empty:
            return _ERROR.format(reason=f"no price data for {ticker} {start}→{end}")
        return df.to_string()

    @with_retry(attempts=3, base_delay=2.0)
    def get_indicators(
        self, ticker: str, indicators: list[str], start: str, end: str
    ) -> str:
        try:
            from stockstats import wrap
        except ImportError:
            return _ERROR.format(reason="stockstats not installed")

        df = yf.download(ticker, start=start, end=end, progress=False, auto_adjust=True)
        if df.empty:
            return _ERROR.format(reason=f"no price data for {ticker}")
        # stockstats expects lowercase column names
        df.columns = [c.lower() for c in df.columns]
        stock = wrap(df)
        rows = {}
        for ind in indicators:
            try:
                rows[ind] = stock[ind]
            except Exception as exc:
                logger.warning("Indicator %r failed: %s", ind, exc)
        if not rows:
            return _ERROR.format(reason="no indicators computed")
        import pandas as pd
        return pd.DataFrame(rows).tail(60).to_string()

    @with_retry(attempts=3, base_delay=2.0)
    def get_fundamentals(self, ticker: str) -> str:
        t = yf.Ticker(ticker)
        info = t.info
        if not info:
            return _ERROR.format(reason=f"no fundamentals for {ticker}")
        keys = [
            "marketCap", "trailingPE", "forwardPE", "priceToBook",
            "returnOnEquity", "debtToEquity", "currentRatio", "quickRatio",
            "revenueGrowth", "earningsGrowth", "grossMargins", "operatingMargins",
        ]
        lines = [f"{k}: {info.get(k, 'N/A')}" for k in keys]
        return "\n".join(lines)

    @with_retry(attempts=3, base_delay=2.0)
    def get_balance_sheet(self, ticker: str) -> str:
        t = yf.Ticker(ticker)
        bs = t.balance_sheet
        if bs is None or bs.empty:
            return _ERROR.format(reason=f"no balance sheet for {ticker}")
        return bs.to_string()

    @with_retry(attempts=3, base_delay=2.0)
    def get_cashflow(self, ticker: str) -> str:
        t = yf.Ticker(ticker)
        cf = t.cashflow
        if cf is None or cf.empty:
            return _ERROR.format(reason=f"no cashflow for {ticker}")
        return cf.to_string()

    @with_retry(attempts=3, base_delay=2.0)
    def get_income_statement(self, ticker: str) -> str:
        t = yf.Ticker(ticker)
        inc = t.income_stmt
        if inc is None or inc.empty:
            return _ERROR.format(reason=f"no income statement for {ticker}")
        return inc.to_string()

    @with_retry(attempts=3, base_delay=2.0)
    def get_news(self, ticker: str, date: str) -> str:
        t = yf.Ticker(ticker)
        news = t.news
        if not news:
            return _ERROR.format(reason=f"no news for {ticker}")
        lines = []
        for item in news[:20]:
            title = item.get("title", "")
            pub = item.get("providerPublishTime", "")
            lines.append(f"[{pub}] {title}")
        return "\n".join(lines)

    @with_retry(attempts=3, base_delay=2.0)
    def get_global_news(self, date: str) -> str:
        # yfinance does not expose a global news endpoint; use SPY proxy
        t = yf.Ticker("SPY")
        news = t.news
        if not news:
            return _ERROR.format(reason="no global news available")
        lines = []
        for item in news[:10]:
            title = item.get("title", "")
            pub = item.get("providerPublishTime", "")
            lines.append(f"[{pub}] {title}")
        return "\n".join(lines)

    @with_retry(attempts=3, base_delay=2.0)
    def get_insider_transactions(self, ticker: str) -> str:
        t = yf.Ticker(ticker)
        ins = t.insider_transactions
        if ins is None or ins.empty:
            return _ERROR.format(reason=f"no insider transactions for {ticker}")
        return ins.to_string()
```

```python
from __future__ import annotations

import logging
import os

from brain_trader.data.vendors.base import BaseDataVendor, with_retry

logger = logging.getLogger(__name__)

_ERROR = "[DATA UNAVAILABLE: {reason}]"
_NO_KEY = _ERROR.format(reason="ALPHA_VANTAGE_API_KEY not set")


def _get_key() -> str | None:
    return os.environ.get("ALPHA_VANTAGE_API_KEY")


class AlphaVantageVendor(BaseDataVendor):
    """Alpha Vantage data vendor.

    Requires ALPHA_VANTAGE_API_KEY env var. Returns structured error strings
    rather than raising when the key is absent or data is unavailable.
    """

    @with_retry(attempts=3, base_delay=2.0)
    def get_stock_data(self, ticker: str, start: str, end: str) -> str:
        key = _get_key()
        if not key:
            return _NO_KEY
        try:
            from alpha_vantage.timeseries import TimeSeries  # type: ignore[import]
            ts = TimeSeries(key=key, output_format="pandas")
            df, _ = ts.get_daily_adjusted(symbol=ticker, outputsize="full")
            df = df[start:end]
            if df.empty:
                return _ERROR.format(reason=f"no price data for {ticker} {start}→{end}")
            return df.to_string()
        except Exception as exc:
            logger.warning("AlphaVantage get_stock_data failed: %s", exc)
            return _ERROR.format(reason=str(exc))

    @with_retry(attempts=3, base_delay=2.0)
    def get_indicators(
        self, ticker: str, indicators: list[str], start: str, end: str
    ) -> str:
        # Alpha Vantage technical indicators are per-function endpoints;
        # fall back to yfinance for indicator computation to avoid N API calls.
        from brain_trader.data.vendors.yfinance import YFinanceVendor
        return YFinanceVendor().get_indicators(ticker, indicators, start, end)

    @with_retry(attempts=3, base_delay=2.0)
    def get_fundamentals(self, ticker: str) -> str:
        key = _get_key()
        if not key:
            return _NO_KEY
        try:
            import requests
            r = requests.get(
                "https://www.alphavantage.co/query",
                params={"function": "OVERVIEW", "symbol": ticker, "apikey": key},
                timeout=15,
            )

            if "Symbol" not in data:
                return _ERROR.format(reason=data.get("Note", "no fundamentals"))

                "MarketCapitalization", "PERatio", "ForwardPE", "PriceToBookRatio",
                "ReturnOnEquityTTM", "DebtToEquityRatio", "CurrentRatio",
                "RevenueGrowthYOY", "GrossProfitTTM", "OperatingMarginTTM",
            ]
            lines = [f"{k}: {data.get(k, 'N/A')}" for k in keys]
            return "\n".join(lines)
        except Exception as exc:
            return _ERROR.format(reason=str(exc))

    @with_retry(attempts=3, base_delay=2.0)
    def get_balance_sheet(self, ticker: str) -> str:
        key = _get_key()
        if not key:
            return _NO_KEY
        try:
            import requests
            r = requests.get(
                "https://www.alphavantage.co/query",
                params={"function": "BALANCE_SHEET", "symbol": ticker, "apikey": key},
                timeout=15,
            )
            data = r.json()
            reports = data.get("annualReports", [])
            if not reports:
                return _ERROR.format(reason=f"no balance sheet for {ticker}")
            import json
            return json.dumps(reports[:2], indent=2)
        except Exception as exc:
            return _ERROR.format(reason=str(exc))
```

```python
# brain_trader/data/tools.py
from __future__ import annotations

from langchain_core.tools import tool

from brain_trader.config.schema import DataVendorConfig
from brain_trader.data.registry import DataVendorRegistry

_registry: DataVendorRegistry | None = None


def configure_registry(config: DataVendorConfig) -> None:
    """Called once by BrainTrader.__init__ to wire up the vendor registry."""
    global _registry
    _registry = DataVendorRegistry(config)


def _get_registry() -> DataVendorRegistry:
    if _registry is None:
        # Fallback: default config — avoids ImportError in tests that don't call configure_registry
        configure_registry(DataVendorConfig())
    return _registry  # type: ignore[return-value]


@tool
def get_stock_data(ticker: str, start: str, end: str) -> str:
    """Fetch OHLCV price data for a ticker between start and end dates (YYYY-MM-DD)."""
    return _get_registry().get_vendor("get_stock_data").get_stock_data(ticker, start, end)


@tool
def get_indicators(ticker: str, indicators: list[str], start: str, end: str) -> str:
    """Compute technical indicators for a ticker. indicators is a list of names like ['rsi', 'macd']."""
    return _get_registry().get_vendor("get_indicators").get_indicators(
        ticker, indicators, start, end
    )


@tool
def get_fundamentals(ticker: str) -> str:
    """Fetch key fundamental metrics (P/E, ROE, debt ratios, etc.) for a ticker."""
    return _get_registry().get_vendor("get_fundamentals").get_fundamentals(ticker)


@tool
def get_balance_sheet(ticker: str) -> str:
    """Fetch the most recent annual balance sheet for a ticker."""
    return _get_registry().get_vendor("get_balance_sheet").get_balance_sheet(ticker)


@tool
def get_cashflow(ticker: str) -> str:
    """Fetch the most recent annual cash flow statement for a ticker."""
    return _get_registry().get_vendor("get_cashflow").get_cashflow(ticker)


@tool
def get_income_statement(ticker: str) -> str:
    """Fetch the most recent annual income statement for a ticker."""
    return _get_registry().get_vendor("get_income_statement").get_income_statement(ticker)


@tool
def get_news(ticker: str, date: str) -> str:
    """Fetch recent news headlines for a ticker. date is the analysis date (YYYY-MM-DD)."""
    return _get_registry().get_vendor("get_news").get_news(ticker, date)


@tool
```
