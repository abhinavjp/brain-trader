**Goal:** Implement the SQLite-backed decision log and the LLM-powered reflector that generates post-trade lessons from realised returns.

**Architecture:** `DecisionLog` owns all reads and writes to a single SQLite file. Decisions are stored as `pending` and resolved on the next same-ticker run once returns are available. `batch_resolve()` updates many rows in one transaction. `get_context()` returns a formatted string injected into the Portfolio Manager prompt. `export_markdown()` produces a human-readable file for inspection. `Reflector` wraps a quick LLM call — given a decision text and raw/alpha returns, it outputs a one-paragraph lesson.

**Tech Stack:** Python 3.12+, stdlib `sqlite3`, langchain-core

**Dependency:** Plan 03 (LLM layer) for `Reflector`. No other plan dependency.

```python
# brain_trader/memory/log.py
from __future__ import annotations

import sqlite3
from datetime import datetime, timezone
from pathlib import Path

_DDL = """
CREATE TABLE IF NOT EXISTS decisions (
    id            INTEGER PRIMARY KEY AUTOINCREMENT,
    ticker        TEXT    NOT NULL,
    trade_date    TEXT    NOT NULL,
    rating        TEXT    NOT NULL,
    decision      TEXT    NOT NULL,
    status        TEXT    NOT NULL DEFAULT 'pending',
    raw_return    REAL,
    alpha_return  REAL,
    holding_days  INTEGER,
    reflection    TEXT,
    created_at    TEXT    NOT NULL
);
CREATE INDEX IF NOT EXISTS idx_ticker_date ON decisions (ticker, trade_date);
CREATE INDEX IF NOT EXISTS idx_status       ON decisions (status);
"""


class DecisionLog:
    """Append-oriented SQLite log of trading decisions and post-trade reflections.

    Lifecycle of a decision:
      1. store()        → status = 'pending'
      2. batch_resolve() → status = 'resolved', returns + reflection written
      3. get_context()  → resolved entries injected into Portfolio Manager prompt
    """

    def __init__(self, db_path: Path) -> None:
        self._path = Path(db_path)
        self._path.parent.mkdir(parents=True, exist_ok=True)
        with self._connect() as conn:
            conn.executescript(_DDL)

    # ------------------------------------------------------------------
    # Write path
    # ------------------------------------------------------------------

    def store(
        self,
        ticker: str,
        trade_date: str,
        rating: str,
        decision: str,
    ) -> None:
        """Append a pending decision. Idempotent — no-op if (ticker, trade_date) pending already."""
        with self._connect() as conn:
            existing = conn.execute(
                "SELECT id FROM decisions WHERE ticker=? AND trade_date=? AND status='pending'",
                (ticker, trade_date),
            ).fetchone()
            if existing:
                return
            conn.execute(
                "INSERT INTO decisions (ticker, trade_date, rating, decision, status, created_at)"
                " VALUES (?, ?, ?, ?, 'pending', ?)",
                (ticker, trade_date, rating, decision, _now()),
            )

    def resolve(
        self,
        ticker: str,
        trade_date: str,
        raw_return: float,
        alpha_return: float,
        holding_days: int,
        reflection: str,
    ) -> None:
        """Resolve a single pending entry."""
        with self._connect() as conn:
            conn.execute(
                "UPDATE decisions"
                " SET status='resolved', raw_return=?, alpha_return=?,"
                "     holding_days=?, reflection=?"
                " WHERE ticker=? AND trade_date=? AND status='pending'",
                (raw_return, alpha_return, holding_days, reflection, ticker, trade_date),
            )

    def batch_resolve(self, updates: list[dict]) -> None:
        """Resolve multiple pending entries in a single transaction.

        Each element must have keys: ticker, trade_date, raw_return,
        alpha_return, holding_days, reflection.
        """
        with self._connect() as conn:
            for u in updates:
                conn.execute(
                    "UPDATE decisions"
                    " SET status='resolved', raw_return=?, alpha_return=?,"
                    "     holding_days=?, reflection=?"
                    " WHERE ticker=? AND trade_date=? AND status='pending'",
                    (
                        u["raw_return"], u["alpha_return"],
                        u["holding_days"], u["reflection"],
                        u["ticker"], u["trade_date"],
                    ),
                )

    # ------------------------------------------------------------------
    # Read path
    # ------------------------------------------------------------------

    def get_pending(self, ticker: str) -> list[dict]:
        """Return all pending entries for a ticker ordered by trade date."""
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT * FROM decisions WHERE ticker=? AND status='pending'"
                " ORDER BY trade_date ASC",
                (ticker,),
            ).fetchall()
        return [dict(r) for r in rows]

    def get_context(
        self,
        ticker: str,
        n_same: int = 5,
        n_cross: int = 3,
    ) -> str:
        """Return a formatted context string for prompt injection.

        Includes the n_same most recent resolved decisions for the same ticker
        and n_cross most recent resolved decisions for other tickers.
        Returns empty string when no resolved entries exist.
        """
        with self._connect() as conn:
            same = conn.execute(
                "SELECT * FROM decisions WHERE ticker=? AND status='resolved'"
                " ORDER BY trade_date DESC LIMIT ?",
                (ticker, n_same),
            ).fetchall()
            cross = conn.execute(
                "SELECT * FROM decisions WHERE ticker!=? AND status='resolved'"
                " ORDER BY trade_date DESC LIMIT ?",
                (ticker, n_cross),
            ).fetchall()

        if not same and not cross:
            return ""

        parts: list[str] = []
        if same:
            parts.append(f"Past analyses of {ticker} (most recent first):")
            parts.extend(_format_full(dict(r)) for r in same)
        if cross:
            parts.append("Recent cross-ticker lessons:")
            parts.extend(_format_reflection_only(dict(r)) for r in cross)
        return "\n\n".join(parts)

    # ------------------------------------------------------------------
    # Maintenance
    # ------------------------------------------------------------------

    def prune(self, max_entries: int) -> None:
        """Delete oldest resolved entries once their count exceeds max_entries.

        Pending entries are never pruned.
        """
        with self._connect() as conn:
            count = conn.execute(
                "SELECT COUNT(*) FROM decisions WHERE status='resolved'"
            ).fetchone()[0]
            excess = count - max_entries
            if excess > 0:
                conn.execute(
                    "DELETE FROM decisions WHERE id IN ("
                    "  SELECT id FROM decisions WHERE status='resolved'"
                    "  ORDER BY created_at ASC LIMIT ?"
                    ")",
                    (excess,),
                )

    def export_markdown(self, path: Path) -> None:
        """Write all entries to a human-readable markdown file."""
        with self._connect() as conn:
            rows = conn.execute(
                "SELECT * FROM decisions ORDER BY created_at ASC"
            ).fetchall()
        separator = "\n\n<!-- ENTRY_END -->\n\n"
        blocks: list[str] = []
        for r in rows:
            d = dict(r)
            if d["status"] == "pending":
                tag = f"[{d['trade_date']} | {d['ticker']} | {d['rating']} | pending]"
                blocks.append(f"{tag}\n\nDECISION:\n{d['decision']}")
            else:
                raw = f"{d['raw_return']:+.1%}" if d["raw_return"] is not None else "n/a"
                alpha = f"{d['alpha_return']:+.1%}" if d["alpha_return"] is not None else "n/a"
                tag = (
                    f"[{d['trade_date']} | {d['ticker']} | {d['rating']}"
                    f" | {raw} | {alpha} | {d['holding_days']}d]"
                )
                entry = f"{tag}\n\nDECISION:\n{d['decision']}"
                if d.get("reflection"):
                    entry += f"\n\nREFLECTION:\n{d['reflection']}"
                blocks.append(entry)
        Path(path).write_text(separator.join(blocks), encoding="utf-8")

    # ------------------------------------------------------------------

    def _connect(self) -> sqlite3.Connection:
        conn = sqlite3.connect(self._path)
        conn.row_factory = sqlite3.Row
        conn.execute("PRAGMA journal_mode=WAL")
        return conn


# ------------------------------------------------------------------
# Formatting helpers
# ------------------------------------------------------------------

def _now() -> str:
    return datetime.now(timezone.utc).isoformat()


def _format_full(d: dict) -> str:
    raw = f"{d['raw_return']:+.1%}" if d["raw_return"] is not None else "n/a"
    alpha = f"{d['alpha_return']:+.1%}" if d["alpha_return"] is not None else "n/a"
    tag = (
        f"[{d['trade_date']} | {d['ticker']} | {d['rating']}"
        f" | {raw} | {alpha} | {d['holding_days']}d]"
    )
    parts = [tag, f"DECISION:\n{d['decision']}"]
    if d.get("reflection"):
        parts.append(f"REFLECTION:\n{d['reflection']}")
    return "\n\n".join(parts)


def _format_reflection_only(d: dict) -> str:
    raw = f"{d['raw_return']:+.1%}" if d["raw_return"] is not None else "n/a"
    tag = f"[{d['trade_date']} | {d['ticker']} | {d['rating']} | {raw}]"
    if d.get("reflection"):
        return f"{tag}\n{d['reflection']}"
    preview = d["decision"][:300]
    suffix = "..." if len(d["decision"]) > 300 else ""
    return f"{tag}\n{preview}{suffix}"
```

```python
# brain_trader/memory/reflection.py
from __future__ import annotations

import logging
from typing import Any

from langchain_core.messages import HumanMessage

logger = logging.getLogger(__name__)

_PROMPT = """You are reviewing a past trading decision to extract a one-paragraph lesson.

Decision made:
{decision}

Outcome after {holding_days} holding days:
- Raw return:   {raw_pct}
- Alpha vs SPY: {alpha_pct}

Write a concise, specific one-paragraph reflection (3-5 sentences) that:
1. States whether the decision was correct given the outcome.
2. Identifies what the decision got right or wrong.
3. Notes one concrete thing to do differently or reinforce next time.

Do not repeat the numbers back verbatim — synthesise them into insight."""


class Reflector:
    """Generates a post-trade reflection paragraph using a quick LLM call."""

    def __init__(self, llm: Any) -> None:
        self._llm = llm

    def reflect(
        self,
        final_decision: str,
        raw_return: float,
        alpha_return: float,
        holding_days: int = 5,
    ) -> str:
        """Return a one-paragraph lesson string.

        Falls back to a formatted summary string if the LLM call fails,
        so memory writes never block on LLM errors.
        """
        prompt = _PROMPT.format(
            decision=final_decision[:2000],
            holding_days=holding_days,
            raw_pct=f"{raw_return:+.2%}",
            alpha_pct=f"{alpha_return:+.2%}",
        )
        try:
            result = self._llm.invoke([HumanMessage(content=prompt)])
            content = result.content if hasattr(result, "content") else str(result)
            return content.strip()
        except Exception as exc:
            logger.warning("Reflector LLM call failed: %s — using fallback.", exc)
            return (
                f"Return: {raw_return:+.2%} (alpha: {alpha_return:+.2%}) "
                f"over {holding_days} days. Reflection unavailable."
            )
```
