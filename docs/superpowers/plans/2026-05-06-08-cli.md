## Task 1 — BrainTrader Core Class

### File: `brain_trader/_core.py`

```python
from __future__ import annotations

from pathlib import Path
from typing import Any

from brain_trader.config.models import BrainTraderConfig
from brain_trader.data.registry import DataVendorRegistry
from brain_trader.graph.builder import GraphBuilder
from brain_trader.graph.runner import GraphRunner, RunResult
from brain_trader.llm.factory import LLMFactory
from brain_trader.memory.log import DecisionLog
from brain_trader.observability.logging import get_logger

_log = get_logger("core")


class BrainTrader:
    """Public Python API for a complete Brain Trader analysis run.

    Usage::

        bt = BrainTrader.from_env()
        result = bt.analyze("AAPL", "2024-01-15")
        print(result.decision)  # BUY / HOLD / SELL
    """

    def __init__(self, config: BrainTraderConfig) -> None:
        self._config = config
        self._registry = DataVendorRegistry(config.data)
        self._deep_llm = LLMFactory.create(config.llm, tier="deep")
        self._quick_llm = LLMFactory.create(config.llm, tier="quick")
        self._memory = DecisionLog(config.memory.db_path)
        graph = GraphBuilder(
            config=config,
            registry=self._registry,
            deep_llm=self._deep_llm,
            quick_llm=self._quick_llm,
            memory=self._memory,
        ).build()
        self._runner = GraphRunner(graph=graph, config=config)

    # ------------------------------------------------------------------
    # Factory
    # ------------------------------------------------------------------

    @classmethod
    def from_env(cls, env_file: Path | None = None) -> "BrainTrader":
        """Construct from environment variables (+ optional .env file)."""
        config = BrainTraderConfig(_env_file=env_file)
        return cls(config)

    @classmethod
    def from_config(cls, config: BrainTraderConfig) -> "BrainTrader":
        """Construct from an explicit config object."""
        return cls(config)

    # ------------------------------------------------------------------
    # Analysis
    # ------------------------------------------------------------------

    def analyze(
        self,
        ticker: str,
        trade_date: str,
        *,
        run_id: str | None = None,
    ) -> RunResult:
        """Run a full analysis for ``ticker`` on ``trade_date``.

        Args:
            ticker: Stock symbol, e.g. ``"AAPL"``.
            trade_date: ISO date string, e.g. ``"2024-01-15"``.
            run_id: Optional stable ID for trace deduplication.

        Returns:
            :class:`RunResult` with decision, full state, token usage, and run_id.
        """
        _log.info("analysis requested", extra={"ticker": ticker, "trade_date": trade_date})
        result = self._runner.run(ticker=ticker, trade_date=trade_date, run_id=run_id)
        # Persist to memory as pending (no return data yet)
        self._memory.store(
            ticker=ticker,
            trade_date=trade_date,
            rating=result.state.get("research_report", ""),
            decision=result.decision,
        )
        return result

    # ------------------------------------------------------------------
    # Memory helpers (pass-through for CLI)
    # ------------------------------------------------------------------

    def pending(self, ticker: str) -> list[dict]:
        return self._memory.get_pending(ticker)

    def resolve(
        self,
        ticker: str,
        trade_date: str,
        *,
        raw_return: float,
        alpha_return: float,
        holding_days: int,
    ) -> None:
        self._memory.resolve(
            ticker=ticker,
            trade_date=trade_date,
            raw_return=raw_return,
            alpha_return=alpha_return,
            holding_days=holding_days,
            reflection="",  # CLI resolve skips LLM reflection
        )

    def export_memory(self, path: Path) -> None:
        self._memory.export_markdown(path)
```

**Rules:**
- `BrainTrader.__init__` builds the entire object graph once — not per `analyze()` call
- `from_env()` is the default constructor for scripts and the CLI
- `from_config()` is the constructor for tests and programmatic use
- `analyze()` stores to memory immediately as pending — resolution happens later via `memory resolve`

## Task 2 — Typer CLI

### File: `brain_trader/cli/main.py`

```python
from __future__ import annotations

import sys
from pathlib import Path
from typing import Annotated, Optional

import typer
from rich.console import Console
from rich.panel import Panel
from rich.table import Table
from rich import box

from brain_trader._core import BrainTrader
from brain_trader.config.models import BrainTraderConfig

app = typer.Typer(
    name="brain-trader",
    help="Multi-agent LLM framework for financial market analysis.",
    no_args_is_help=True,
)
console = Console()
err_console = Console(stderr=True, style="bold red")

# ---------------------------------------------------------------------------
# analyze
# ---------------------------------------------------------------------------


@app.command()
def analyze(
    ticker: Annotated[str, typer.Argument(help="Stock symbol, e.g. AAPL")],
    date: Annotated[str, typer.Argument(help="Trade date ISO string, e.g. 2024-01-15")],
    env_file: Annotated[Optional[Path], typer.Option("--env", help=".env file path")] = None,
    quiet: Annotated[bool, typer.Option("--quiet", "-q", help="Only print decision")] = False,
) -> None:
    """Run a full multi-agent analysis for TICKER on DATE."""
    try:
        bt = BrainTrader.from_env(env_file=env_file)
        with console.status(f"[cyan]Analyzing {ticker} for {date}...[/cyan]"):
            result = bt.analyze(ticker, date)
    except Exception as exc:
        err_console.print(f"Analysis failed: {exc}")
        raise typer.Exit(code=1)

    if quiet:
        console.print(result.decision)
        return

    _render_result(result)


def _render_result(result: "RunResult") -> None:  # type: ignore[name-defined]
    colour = {"BUY": "green", "SELL": "red", "HOLD": "yellow"}.get(result.decision, "white")

    console.print(
        Panel(
            f"[bold {colour}]{result.decision}[/bold {colour}]",
            title=f"[bold]{result.ticker}[/bold] — {result.trade_date}",
            subtitle=f"run_id: {result.run_id}",
            box=box.DOUBLE,
            expand=False,
        )
    )

    # Token usage table
    u = result.usage
    table = Table(box=box.SIMPLE, show_header=True, header_style="bold cyan")
    table.add_column("Metric", style="dim")
    table.add_column("Value", justify="right")
    table.add_row("Prompt tokens", str(u.prompt_tokens))
    table.add_row("Completion tokens", str(u.completion_tokens))
    table.add_row("Total tokens", str(u.total_tokens))
    table.add_row("Estimated cost", f"${u.estimated_usd:.4f}")
    table.add_row("LLM calls", str(u.calls))
    console.print(table)

    # Key reports (collapsed)
    for label, key in [
        ("Market Report", "market_report"),
        ("Research Report", "research_report"),
        ("Trade Proposal", "trade_proposal"),
        ("Final Decision", "final_decision"),
    ]:
        value = result.state.get(key, "")
        if value:
            console.print(Panel(value[:400] + ("…" if len(value) > 400 else ""),
                                title=label, box=box.ROUNDED, expand=False))


# ---------------------------------------------------------------------------
# memory
# ---------------------------------------------------------------------------

memory_app = typer.Typer(help="Inspect and manage decision memory.")
app.add_typer(memory_app, name="memory")


@memory_app.command("pending")
def memory_pending(
    ticker: Annotated[str, typer.Argument(help="Stock symbol")],
    env_file: Annotated[Optional[Path], typer.Option("--env")] = None,
) -> None:
    """List pending (unresolved) decisions for TICKER."""
    bt = BrainTrader.from_env(env_file=env_file)
    rows = bt.pending(ticker)
    if not rows:
        console.print(f"[dim]No pending decisions for {ticker}[/dim]")
        return
    table = Table(title=f"Pending — {ticker}", box=box.SIMPLE)
    table.add_column("trade_date")
    table.add_column("decision")
    table.add_column("created_at")
    for row in rows:
        table.add_row(row["trade_date"], row["decision"], row["created_at"])
    console.print(table)


@memory_app.command("resolve")
def memory_resolve(
    ticker: Annotated[str, typer.Argument()],
    trade_date: Annotated[str, typer.Argument()],
    raw_return: Annotated[float, typer.Option("--raw", help="Actual % return")],
    alpha: Annotated[float, typer.Option("--alpha", help="Alpha vs benchmark")] = 0.0,
    days: Annotated[int, typer.Option("--days", help="Holding period in days")] = 5,
    env_file: Annotated[Optional[Path], typer.Option("--env")] = None,
) -> None:
    """Mark a pending decision as resolved with actual return data."""
    bt = BrainTrader.from_env(env_file=env_file)
    bt.resolve(
        ticker,
        trade_date,
        raw_return=raw_return,
        alpha_return=alpha,
        holding_days=days,
    )
    console.print(f"[green]Resolved {ticker} {trade_date}[/green]")


@memory_app.command("export")
def memory_export(
    path: Annotated[Path, typer.Argument(help="Output .md file path")],
    env_file: Annotated[Optional[Path], typer.Option("--env")] = None,
) -> None:
    """Export full decision log to a Markdown file."""
    bt = BrainTrader.from_env(env_file=env_file)
    bt.export_memory(path)
    console.print(f"[green]Exported to {path}[/green]")


# ---------------------------------------------------------------------------
# config
# ---------------------------------------------------------------------------

config_app = typer.Typer(help="Configuration utilities.")
app.add_typer(config_app, name="config")


@config_app.command("show")
def config_show(
    env_file: Annotated[Optional[Path], typer.Option("--env")] = None,
) -> None:
    """Print the active configuration (secrets redacted)."""
    config = BrainTraderConfig(_env_file=env_file)
    dumped = config.model_dump()
    _redact(dumped)

    import json
    console.print_json(json.dumps(dumped, default=str))


@config_app.command("wizard")
def config_wizard() -> None:
    """Interactive wizard to create a .env file."""
    console.print("[bold cyan]Brain Trader — Configuration Wizard[/bold cyan]")
    lines: list[str] = []

    def _ask(prompt: str, default: str = "", secret: bool = False) -> str:
        if secret:
            return typer.prompt(prompt, default=default, hide_input=True)
        return typer.prompt(prompt, default=default)

    lines.append(f'BRAIN_TRADER_LLM__PROVIDER={_ask("LLM provider", "openai")}')
    lines.append(f'BRAIN_TRADER_LLM__DEEP_MODEL={_ask("Deep model", "gpt-4o")}')
    lines.append(f'BRAIN_TRADER_LLM__QUICK_MODEL={_ask("Quick model", "gpt-4o-mini")}')
    lines.append(f'OPENAI_API_KEY={_ask("OpenAI API key", secret=True)}')

    add_av = typer.confirm("Add Alpha Vantage key?", default=False)
    if add_av:
        lines.append(f'ALPHA_VANTAGE_API_KEY={_ask("Alpha Vantage API key", secret=True)}')

    env_path = Path(".env")
    env_path.write_text("\n".join(lines) + "\n")
    console.print(f"[green].env written to {env_path.resolve()}[/green]")


def _redact(d: dict, secrets: tuple[str, ...] = ("key", "token", "secret", "password")) -> None:
    for k, v in list(d.items()):
        if isinstance(v, dict):
            _redact(v, secrets)
        elif isinstance(v, str) and any(s in k.lower() for s in secrets):
            d[k] = "***"


# ---------------------------------------------------------------------------
# Entry point
# ---------------------------------------------------------------------------

def main() -> None:  # pyproject.toml entry_point
    app()
```

## Notes

- `config wizard` writes `.env` interactively — it never prints secrets to stdout
- `_redact()` ensures `config show` never leaks API keys
- `analyze --quiet` is designed for scripting: `DECISION=$(brain-trader analyze AAPL 2024-01-15 --quiet)`
- All subcommands accept `--env PATH` to point at a non-default `.env` file
- `BrainTrader` is the single import users need from the library surface
