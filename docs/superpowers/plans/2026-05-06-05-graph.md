```python
# brain_trader/graph/signals.py
from __future__ import annotations

import re

# Ordered by specificity — STRONG variants checked before plain BUY/SELL
_PATTERNS: list[tuple[str, str]] = [
    ("STRONG BUY",  r"\bSTRONG[\s_]BUY\b"),
    ("STRONG SELL", r"\bSTRONG[\s_]SELL\b"),
    ("BUY",  r"\b(?:FINAL\s+)?(?:TRANSACTION\s+PROPOSAL\s*:\s*\*{0,2}\s*)?BUY\b"),
    ("SELL", r"\b(?:FINAL\s+)?(?:(?:TRANSACTION\s+PROPOSAL|DECISION)\s*:\s*\*{0,2}\s*)?SELL\b"),
    ("HOLD", r"\b(?:FINAL\s+)?(?:(?:TRANSACTION\s+PROPOSAL|DECISION)\s*:\s*\*{0,2}\s*)?HOLD\b"),
]


def extract_signal(text: str) -> str:
    """Extract a BUY/HOLD/SELL signal from verbose decision text.

    Tries each pattern in priority order. Defaults to HOLD when nothing matches
    so downstream code always gets a valid signal.
    """
    upper = text.upper()
    for label, pattern in _PATTERNS:
        if re.search(pattern, upper):
            return label
    return "HOLD"


def parse_rating(text: str) -> str:
    """Alias for extract_signal — used when storing rating in the decision log."""
    return extract_signal(text)
```

```python
# brain_trader/graph/conditions.py
from __future__ import annotations

# Each function is a LangGraph conditional edge callback.
# State dict keys prefixed with "_" are injected by GraphBuilder at compile time
# via functools.partial so the functions stay pure and testable.

# ---------------------------------------------------------------------------
# Analyst continue conditions
# One per analyst type — LangGraph requires distinct callables per edge.
# ---------------------------------------------------------------------------

def _analyst_condition(state: dict, tools_node: str, clear_node: str) -> str:
    """Route to tool node if last message has tool calls, else to clear node."""
    messages = state.get("messages", [])
    if messages and getattr(messages[-1], "tool_calls", None):
        return tools_node
    return clear_node


def should_continue_market(state: dict) -> str:
    return _analyst_condition(state, "tools_market", "Msg Clear Market")


def should_continue_news(state: dict) -> str:
    return _analyst_condition(state, "tools_news", "Msg Clear News")


def should_continue_social(state: dict) -> str:
    return _analyst_condition(state, "tools_social", "Msg Clear Social")


def should_continue_fundamentals(state: dict) -> str:
    return _analyst_condition(state, "tools_fundamentals", "Msg Clear Fundamentals")


# ---------------------------------------------------------------------------
# Research debate condition
# ---------------------------------------------------------------------------

def should_continue_debate(state: dict) -> str:
    """Route between Bull/Bear debate and Research Manager.

    Reads _max_debate_rounds injected by GraphBuilder, or falls back to
    the config value stored in state.
    """
    ds = state.get("investment_debate_state", {})
    count = ds.get("count", 0)
    max_rounds = state.get("_max_debate_rounds", 1)
    # Bull fires first, so even counts mean bear has spoken — check against 2*max
    if count >= max_rounds * 2:
        return "Research Manager"
    # Alternate: bull goes on odd counts, bear on even
    if count % 2 == 1:
        return "Bear Researcher"
    return "Research Manager"


# ---------------------------------------------------------------------------
# Risk debate condition
# ---------------------------------------------------------------------------

_RISK_ROTATION: dict[str, str] = {
    "aggressive": "Conservative Analyst",
    "conservative": "Neutral Analyst",
    "neutral": "Aggressive Analyst",
}


def should_continue_risk(state: dict) -> str:
    """Rotate through risk debaters until max rounds, then go to Portfolio Manager."""
    rds = state.get("risk_debate_state", {})
    count = rds.get("count", 0)
    max_rounds = state.get("_max_risk_rounds", 1)
    if count >= max_rounds * 3:  # 3 debaters per round
        return "Portfolio Manager"
    latest = rds.get("latest_speaker", "neutral")
    return _RISK_ROTATION.get(latest, "Portfolio Manager")
```

```python
# brain_trader/graph/builder.py
from __future__ import annotations

from typing import Any

from langchain_core.language_models import BaseChatModel
from langchain_core.tools import BaseTool
from langgraph.graph import END, START, StateGraph
from langgraph.prebuilt import ToolNode

from brain_trader.agents import (
    create_aggressive_debater,
    create_bear_researcher,
    create_bull_researcher,
    create_conservative_debater,
    create_fundamentals_analyst,
    create_market_analyst,
    create_msg_delete,
    create_news_analyst,
    create_neutral_debater,
    create_portfolio_manager,
    create_research_manager,
    create_social_analyst,
    create_trader,
)
from brain_trader.agents.state import AgentState
from brain_trader.config.schema import BrainTraderConfig
from brain_trader.data.tools import (
    get_balance_sheet,
    get_cashflow,
    get_fundamentals,
    get_income_statement,
    get_indicators,
    get_insider_transactions,
    get_global_news,
    get_news,
    get_stock_data,
)
from brain_trader.graph.conditions import (
    should_continue_debate,
    should_continue_fundamentals,
    should_continue_market,
    should_continue_news,
    should_continue_risk,
    should_continue_social,
)

# Maps analyst name → (condition fn, tool list)
_ANALYST_REGISTRY: dict[str, tuple[Any, list[BaseTool]]] = {
    "market":       (should_continue_market,       [get_stock_data, get_indicators]),
    "news":         (should_continue_news,          [get_news, get_global_news, get_insider_transactions]),
    "social":       (should_continue_social,        [get_news]),
    "fundamentals": (should_continue_fundamentals,  [get_fundamentals, get_balance_sheet,
                                                     get_cashflow, get_income_statement]),
}

# Maps analyst name → node factory
_ANALYST_FACTORIES: dict[str, Any] = {
    "market":       create_market_analyst,
    "news":         create_news_analyst,
    "social":       create_social_analyst,
    "fundamentals": create_fundamentals_analyst,
}

# Node name templates
_ANALYST_NODE  = "{name} Analyst"
_TOOLS_NODE    = "tools_{key}"
_CLEAR_NODE    = "Msg Clear {name}"


class GraphBuilder:
    """Constructs a compiled LangGraph StateGraph from config and LLM objects.

    Keeps zero execution state — safe to call build() multiple times.
    """

    def __init__(
        self,
        config: BrainTraderConfig,
        quick_llm: BaseChatModel,
        deep_llm: BaseChatModel,
    ) -> None:
        self._config = config
        self._quick = quick_llm
        self._deep = deep_llm

    def build(self) -> Any:
        """Return a compiled StateGraph ready for invoke() / stream()."""
        selected = self._config.analysts
        if not selected:
            raise ValueError("At least one analyst must be selected.")

        workflow = StateGraph(AgentState)

        # --- Analyst phase nodes ---
        for key in selected:
            name = key.capitalize()
            factory = _ANALYST_FACTORIES[key]
            condition_fn, tools = _ANALYST_REGISTRY[key]

            workflow.add_node(_ANALYST_NODE.format(name=name), factory(self._quick))
            workflow.add_node(_TOOLS_NODE.format(key=key), ToolNode(tools))
            workflow.add_node(_CLEAR_NODE.format(name=name), create_msg_delete())

        # --- Researcher + manager + trader nodes ---
        workflow.add_node("Bull Researcher",   create_bull_researcher(self._quick))
        workflow.add_node("Bear Researcher",   create_bear_researcher(self._quick))
        workflow.add_node("Research Manager",  create_research_manager(self._deep))
        workflow.add_node("Trader",            create_trader(self._quick))

        # --- Risk panel + portfolio manager nodes ---
        workflow.add_node("Aggressive Analyst",  create_aggressive_debater(self._quick))
        workflow.add_node("Conservative Analyst", create_conservative_debater(self._quick))
        workflow.add_node("Neutral Analyst",      create_neutral_debater(self._quick))
        workflow.add_node("Portfolio Manager",    create_portfolio_manager(self._deep))

        # --- Edges: analyst phase ---
        first = selected[0].capitalize()
        workflow.add_edge(START, _ANALYST_NODE.format(name=first))

        for i, key in enumerate(selected):
            name = key.capitalize()
            condition_fn, _ = _ANALYST_REGISTRY[key]

            workflow.add_conditional_edges(
                _ANALYST_NODE.format(name=name),
                condition_fn,
                [_TOOLS_NODE.format(key=key), _CLEAR_NODE.format(name=name)],
            )
            workflow.add_edge(_TOOLS_NODE.format(key=key), _ANALYST_NODE.format(name=name))

            if i < len(selected) - 1:
                next_name = selected[i + 1].capitalize()
                workflow.add_edge(
                    _CLEAR_NODE.format(name=name),
                    _ANALYST_NODE.format(name=next_name),
                )
            else:
                workflow.add_edge(_CLEAR_NODE.format(name=name), "Bull Researcher")

        # --- Edges: research debate ---
        workflow.add_conditional_edges(
            "Bull Researcher",
            should_continue_debate,
            {"Bear Researcher": "Bear Researcher", "Research Manager": "Research Manager"},
        )
        workflow.add_conditional_edges(
            "Bear Researcher",
            should_continue_debate,
            {"Bull Researcher": "Bull Researcher", "Research Manager": "Research Manager"},
        )

        # --- Edges: trader ---
        workflow.add_edge("Research Manager", "Trader")

        # --- Edges: risk phase ---
        workflow.add_edge("Trader", "Aggressive Analyst")
        workflow.add_conditional_edges(
            "Aggressive Analyst",
            should_continue_risk,
            {"Conservative Analyst": "Conservative Analyst", "Portfolio Manager": "Portfolio Manager"},
        )
        workflow.add_conditional_edges(
            "Conservative Analyst",
            should_continue_risk,
            {"Neutral Analyst": "Neutral Analyst", "Portfolio Manager": "Portfolio Manager"},
        )
        workflow.add_conditional_edges(
            "Neutral Analyst",
            should_continue_risk,
            {"Aggressive Analyst": "Aggressive Analyst", "Portfolio Manager": "Portfolio Manager"},
        )
        workflow.add_edge("Portfolio Manager", END)

        return workflow.compile()
```

```python
# brain_trader/graph/runner.py
from __future__ import annotations

import json
import logging
import time
from pathlib import Path
from typing import Any

from langchain_core.messages import HumanMessage

from brain_trader.config.schema import BrainTraderConfig
from brain_trader.graph.signals import extract_signal

logger = logging.getLogger(__name__)


class GraphRunner:
    """Executes a compiled LangGraph graph for a single ticker + date.

    Responsibilities:
    - Build initial AgentState
    - Stream or invoke the graph
    - Write the per-run trace JSON
    - Return (final_state, signal)

    Does NOT manage checkpoints directly — the caller (BrainTrader) compiles
    the graph with or without a checkpointer before passing it in.
    """

    def __init__(self, config: BrainTraderConfig) -> None:
        self._config = config

    def run(
        self,
        graph: Any,
        ticker: str,
        date: str,
        past_context: str,
        run_id: str,
        thread_id: str | None = None,
    ) -> tuple[dict, str]:
        """Execute the graph and return (final_state, signal).

        Args:
            graph:        Compiled StateGraph (with or without checkpointer).
            ticker:       Ticker symbol.
            date:         Analysis date string (YYYY-MM-DD).
            past_context: Memory context injected into Portfolio Manager prompt.
            run_id:       UUID for this run — used in trace file name.
            thread_id:    LangGraph thread ID for checkpoint resume (optional).
        """
        init_state = self._build_initial_state(ticker, date, past_context)
        invoke_kwargs: dict[str, Any] = {
            "recursion_limit": self._config.max_recur_limit,
        }
        if thread_id:
            invoke_kwargs["config"] = {"configurable": {"thread_id": thread_id}}

        start = time.monotonic()
        final_state = graph.invoke(init_state, **invoke_kwargs)
        elapsed = time.monotonic() - start

        signal = extract_signal(final_state.get("final_trade_decision", ""))
        self._write_trace(ticker, date, run_id, elapsed, final_state, signal)
        logger.info(
            "run_id=%s ticker=%s date=%s signal=%s elapsed=%.2fs",
            run_id, ticker, date, signal, elapsed,
        )
        return final_state, signal

    # ------------------------------------------------------------------

    def _build_initial_state(self, ticker: str, date: str, past_context: str) -> dict:
        return {
            "messages": [HumanMessage(content=f"Analyse {ticker} for {date}.")],
            "company_of_interest": ticker,
            "trade_date": date,
            "sender": "",
            "market_report": "",
            "sentiment_report": "",
            "news_report": "",
            "fundamentals_report": "",
            "investment_debate_state": {
                "bull_history": "", "bear_history": "", "history": "",
                "current_response": "", "judge_decision": "", "count": 0,
            },
            "investment_plan": "",
            "trader_investment_plan": "",
            "risk_debate_state": {
                "aggressive_history": "", "conservative_history": "",
                "neutral_history": "", "history": "", "latest_speaker": "",
                "current_aggressive_response": "", "current_conservative_response": "",
                "current_neutral_response": "", "judge_decision": "", "count": 0,
            },
            "final_trade_decision": "",
            "past_context": past_context,
            # Injected for conditional edge functions
            "_max_debate_rounds": self._config.max_debate_rounds,
            "_max_risk_rounds": self._config.max_risk_rounds,
        }

    def _write_trace(
        self,
        ticker: str,
        date: str,
        run_id: str,
        elapsed: float,
        final_state: dict,
        signal: str,
    ) -> None:
        if not self._config.observability.trace_enabled:
            return
        trace_dir = self._config.results_dir / _safe(ticker) / "traces"
        trace_dir.mkdir(parents=True, exist_ok=True)
        trace = {
            "run_id": run_id,
            "ticker": ticker,
            "date": date,
            "signal": signal,
            "elapsed_seconds": round(elapsed, 3),
            "reports": {
                "market":       final_state.get("market_report", "")[:500],
                "sentiment":    final_state.get("sentiment_report", "")[:500],
                "news":         final_state.get("news_report", "")[:500],
                "fundamentals": final_state.get("fundamentals_report", "")[:500],
            },
            "final_decision": final_state.get("final_trade_decision", "")[:1000],
        }
        path = trace_dir / f"{run_id}.json"
        with open(path, "w", encoding="utf-8") as f:
            json.dump(trace, f, indent=2)


def _safe(ticker: str) -> str:
    """Sanitise ticker for use as a directory component."""
    return "".join(c if c.isalnum() or c in "-_." else "_" for c in ticker)
```
