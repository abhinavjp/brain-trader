# Brain Trader — Plan 03: LLM Layer

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement the LLM provider abstraction — a factory that returns a LangChain-compatible LLM object for any supported provider, driven entirely by config.

**Architecture:** `BaseLLMClient` defines `get_llm() -> BaseChatModel`. Five concrete clients cover all providers: `OpenAIClient` (also handles xAI/DeepSeek/Qwen/GLM/Ollama/OpenRouter via base URL swapping), `AnthropicClient`, `GoogleClient`, `AzureClient`. `ModelCatalog` maps model names to providers for validation. `LLMFactory.create()` is the single entry point used by `BrainTrader.__init__`.

**Tech Stack:** Python 3.12+, langchain-openai, langchain-anthropic, langchain-google-genai, pydantic v2

```python
# brain_trader/llm/catalog.py
from __future__ import annotations

# Maps model-name prefixes to their canonical provider string.
# Entries are checked with str.startswith() in order — put more specific
# prefixes before broader ones.
_PREFIX_TO_PROVIDER: list[tuple[str, str]] = [
    ("gpt-",          "openai"),
    ("o1",            "openai"),
    ("o3",            "openai"),
    ("o4",            "openai"),
    ("claude-",       "anthropic"),
    ("gemini-",       "google"),
    ("grok-",         "xai"),
    ("deepseek-",     "deepseek"),
    ("qwen-",         "qwen"),
    ("glm-",          "glm"),
]

_ALL_PROVIDERS = frozenset({
    "openai", "anthropic", "google", "azure",
    "xai", "deepseek", "qwen", "glm", "ollama", "openrouter",
})


class ModelCatalog:
    """Lightweight lookup: model name → canonical provider string."""

    @staticmethod
    def provider_for(model: str) -> str | None:
        """Return the provider for a model name, or None if unknown."""
        lower = model.lower()
        for prefix, provider in _PREFIX_TO_PROVIDER:
            if lower.startswith(prefix):
                return provider
        return None

    @staticmethod
    def known_providers() -> frozenset[str]:
        return _ALL_PROVIDERS
```

```python
from __future__ import annotations

from typing import Any

from langchain_core.language_models import BaseChatModel
from langchain_openai import ChatOpenAI

from brain_trader.llm.providers.base import BaseLLMClient


class OpenAIClient(BaseLLMClient):
    def __init__(
        self,
        model: str,
        base_url: str | None = None,
        reasoning_effort: str | None = None,
        callbacks: list[Any] | None = None,
        **kwargs: Any,
    ) -> None:
        self._model = model
        self._base_url = base_url
        self._reasoning_effort = reasoning_effort
        self._callbacks = callbacks or []
        self._extra = kwargs

    def get_llm(self) -> BaseChatModel:
        model_kwargs: dict[str, Any] = {}
        if self._reasoning_effort:
            model_kwargs["reasoning_effort"] = self._reasoning_effort

        return ChatOpenAI(
            model=self._model,
            base_url=self._base_url,
            model_kwargs=model_kwargs or None,
            callbacks=self._callbacks or None,
            **self._extra,
        )
```

```python
from __future__ import annotations

from typing import Any

from langchain_core.language_models import BaseChatModel
from langchain_openai import ChatOpenAI

from brain_trader.llm.providers.base import BaseLLMClient

# Default base URLs for each OpenAI-compatible provider.
# Override via LLMConfig.backend_url if needed.
_BASE_URLS: dict[str, str] = {
    "xai":        "https://api.x.ai/v1",
    "deepseek":   "https://api.deepseek.com/v1",
    "qwen":       "https://dashscope.aliyuncs.com/compatible-mode/v1",
    "glm":        "https://open.bigmodel.cn/api/paas/v4",
    "ollama":     "http://localhost:11434/v1",
    "openrouter": "https://openrouter.ai/api/v1",
}

# Env var names for each provider's API key.
_KEY_ENV: dict[str, str] = {
    "xai":        "XAI_API_KEY",
    "deepseek":   "DEEPSEEK_API_KEY",
    "qwen":       "DASHSCOPE_API_KEY",
    "glm":        "ZHIPU_API_KEY",
    "ollama":     "OLLAMA_API_KEY",   # ollama doesn't require a key; placeholder
    "openrouter": "OPENROUTER_API_KEY",
}


class OpenAICompatClient(BaseLLMClient):
    def __init__(
        self,
        model: str,
        provider: str,
        api_key: str | None = None,
        base_url: str | None = None,
        callbacks: list[Any] | None = None,
        **kwargs: Any,
    ) -> None:
        self._model = model
        self._provider = provider.lower()
        self._api_key = api_key
        self._base_url = base_url or _BASE_URLS.get(self._provider)
        self._callbacks = callbacks or []
        self._extra = kwargs

    def get_llm(self) -> BaseChatModel:
        import os
        api_key = self._api_key or os.environ.get(
            _KEY_ENV.get(self._provider, "OPENAI_API_KEY"), "nokey"
        )
        return ChatOpenAI(
            model=self._model,
            openai_api_key=api_key,
            openai_api_base=self._base_url,
            callbacks=self._callbacks or None,
            **self._extra,
        )
```

```python
# brain_trader/llm/providers/anthropic.py
from __future__ import annotations

from typing import Any

from langchain_anthropic import ChatAnthropic
from langchain_core.language_models import BaseChatModel

from brain_trader.llm.providers.base import BaseLLMClient

# Maps effort strings to approximate token budgets for extended thinking.
_EFFORT_TOKENS: dict[str, int] = {
    "low":    1_000,
    "medium": 5_000,
    "high":   10_000,
}


class AnthropicClient(BaseLLMClient):
    def __init__(
        self,
        model: str,
        base_url: str | None = None,
        effort: str | None = None,
        callbacks: list[Any] | None = None,
        **kwargs: Any,
    ) -> None:
        self._model = model
        self._base_url = base_url
        self._effort = effort
        self._callbacks = callbacks or []
        self._extra = kwargs

    def get_llm(self) -> BaseChatModel:
        model_kwargs: dict[str, Any] = {}
        if self._effort and self._effort.lower() in _EFFORT_TOKENS:
            model_kwargs["thinking"] = {
                "type": "enabled",
                "budget_tokens": _EFFORT_TOKENS[self._effort.lower()],
            }

        return ChatAnthropic(
            model=self._model,
            base_url=self._base_url,
            model_kwargs=model_kwargs or None,
            callbacks=self._callbacks or None,
            **self._extra,
        )
```

```python
# brain_trader/llm/factory.py
from __future__ import annotations

from typing import Any, Literal

from langchain_core.language_models import BaseChatModel

from brain_trader.config.schema import LLMConfig
from brain_trader.llm.providers.base import BaseLLMClient

_OPENAI_COMPAT = frozenset({"xai", "deepseek", "qwen", "glm", "ollama", "openrouter"})


class LLMFactory:
    """Single entry point for creating LangChain LLM objects from config."""

    @staticmethod
    def create(
        config: LLMConfig,
        tier: Literal["deep", "quick"],
        callbacks: list[Any] | None = None,
    ) -> BaseChatModel:
        """Return a configured LangChain chat model.

        Args:
            config:    LLMConfig from BrainTraderConfig.
            tier:      "deep" uses config.deep_model; "quick" uses config.quick_model.
            callbacks: Optional LangChain callbacks (e.g. CostTracker).
        """
        model = config.deep_model if tier == "deep" else config.quick_model
        provider = config.provider.lower()
        client = LLMFactory._build_client(provider, model, config, callbacks)
        return client.get_llm()

    @staticmethod
    def _build_client(
        provider: str,
        model: str,
        config: LLMConfig,
        callbacks: list[Any] | None,
    ) -> BaseLLMClient:
        kw: dict[str, Any] = {"callbacks": callbacks}

        if provider == "openai":
            from brain_trader.llm.providers.openai import OpenAIClient
            return OpenAIClient(
                model=model,
                base_url=config.backend_url,
                reasoning_effort=config.reasoning_effort,
                **kw,
            )

        if provider in _OPENAI_COMPAT:
            from brain_trader.llm.providers.openai_compat import OpenAICompatClient
            return OpenAICompatClient(
                model=model,
                provider=provider,
                base_url=config.backend_url,
                **kw,
            )

        if provider == "anthropic":
            from brain_trader.llm.providers.anthropic import AnthropicClient
            return AnthropicClient(
                model=model,
                base_url=config.backend_url,
                effort=config.effort,
                **kw,
            )

        if provider == "google":
            from brain_trader.llm.providers.google import GoogleClient
            return GoogleClient(
                model=model,
                thinking_level=config.thinking_level,
                **kw,
            )

        if provider == "azure":
            from brain_trader.llm.providers.azure import AzureClient
            return AzureClient(
                model=model,
                base_url=config.backend_url,
                **kw,
            )

        raise ValueError(
            f"Unsupported provider: {provider!r}. "
            f"Choose from: openai, anthropic, google, azure, "
            f"xai, deepseek, qwen, glm, ollama, openrouter."
        )
```
