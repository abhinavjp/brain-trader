> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement `AgentState`, shared prompt utilities, and all 11 agent node functions (4 analysts, 2 researchers, 3 risk agents, 2 managers, 1 trader) plus the message-clearing utility node.

**Architecture:** Every agent is a factory function `create_X(llm) -> Callable[[AgentState], dict]` that returns a LangGraph node. Analysts use LangChain tool-binding and write named reports to state. Researchers/risk agents read accumulated state and write to nested debate state dicts. Managers and the trader use deep-think LLM and write structured decisions. All system prompts live inline in each agent file — no shared prompt templates that obscure intent.

**Tech Stack:** Python 3.12+, langgraph, langchain-core, langchain-core prompts

**Dependency:** Plan 02 (data tools) and Plan 03 (LLM factory) must be complete — agents import from `brain_trader.data.tools`.

from __future__ import annotations

from typing import Annotated
from typing_extensions import TypedDict
from langgraph.graph import MessagesState


class InvestDebateState(TypedDict):
    bull_history: Annotated[str, "Bullish argument history"]
    bear_history: Annotated[str, "Bearish argument history"]
    history: Annotated[str, "Full debate transcript"]
    current_response: Annotated[str, "Latest response in the debate"]
    judge_decision: Annotated[str, "Research Manager's synthesis"]
    count: Annotated[int, "Number of debate rounds completed"]


class RiskDebateState(TypedDict):
    aggressive_history: Annotated[str, "Aggressive debater history"]
    conservative_history: Annotated[str, "Conservative debater history"]
    neutral_history: Annotated[str, "Neutral debater history"]
    history: Annotated[str, "Full risk debate transcript"]
    latest_speaker: Annotated[str, "Name of the last agent to speak"]
    current_aggressive_response: Annotated[str, "Latest aggressive response"]
    current_conservative_response: Annotated[str, "Latest conservative response"]
    current_neutral_response: Annotated[str, "Latest neutral response"]
    judge_decision: Annotated[str, "Portfolio Manager's final ruling"]
    count: Annotated[int, "Number of risk debate rounds completed"]

    company_of_interest: Annotated[str, "Ticker or company under analysis"]
    trade_date: Annotated[str, "Analysis date (YYYY-MM-DD)"]

# brain_trader/agents/_prompts.py
from __future__ import annotations

_ANALYST_SYSTEM_PREFIX = (
    "You are a helpful AI assistant collaborating with other assistants."
    " Use the provided tools to progress towards answering the question."
    " If you are unable to fully answer, that's OK — another assistant with different tools"
    " will help where you left off. Execute what you can to make progress."
    " If you or any other assistant has the FINAL TRANSACTION PROPOSAL:"
    " **BUY/HOLD/SELL** or deliverable,"
    " prefix your response with FINAL TRANSACTION PROPOSAL: **BUY/HOLD/SELL**"
    " so the team knows to stop."
    " You have access to the following tools: {tool_names}.\n{system_message}"
    " For your reference, the current date is {current_date}. {instrument_context}"
)


def get_language_instruction(language: str = "English") -> str:
    """Return an instruction to respond in a non-English language, or empty string."""
    if language.strip().lower() == "english":
        return ""
    return f" Please provide your entire analysis in {language}."


def build_instrument_context(ticker: str) -> str:
    return f"The instrument under analysis is: {ticker}."


def analyst_system_prefix() -> str:
    """Return the shared system prefix template for tool-using analyst agents."""
    return _ANALYST_SYSTEM_PREFIX
```

# brain_trader/agents/analysts/market.py
from __future__ import annotations

from typing import Any, Callable

from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder

from brain_trader.agents._prompts import (
    analyst_system_prefix,
    build_instrument_context,
    get_language_instruction,
)
from brain_trader.data.tools import get_indicators, get_stock_data

_SYSTEM = """You are a market and technical analyst. Your role is to select the most relevant
indicators for the current market condition (up to 8) from the list below, avoiding redundancy,
then produce a detailed, nuanced technical report with specific actionable insights.

Available indicators:
Moving averages: close_50_sma, close_200_sma, close_10_ema
MACD: macd, macds, macdh
Momentum: rsi
Volatility: boll, boll_ub, boll_lb, atr
Volume: vwma

Workflow:
1. Call get_stock_data first to retrieve the price CSV needed for indicator computation.
2. Call get_indicators with your chosen indicator names.
3. Write your report. Append a Markdown summary table at the end.{language}"""


def create_market_analyst(llm: Any) -> Callable[[dict], dict]:
    tools = [get_stock_data, get_indicators]
    tool_names = ", ".join(t.name for t in tools)

    prompt = ChatPromptTemplate.from_messages([
        ("system", analyst_system_prefix()),
        MessagesPlaceholder(variable_name="messages"),
    ])

    def market_analyst_node(state: dict) -> dict:
        system_message = _SYSTEM.format(
            language=get_language_instruction()
        )
        filled = prompt.partial(
            system_message=system_message,
            tool_names=tool_names,
            current_date=state["trade_date"],
            instrument_context=build_instrument_context(state["company_of_interest"]),
        )
        chain = filled | llm.bind_tools(tools)
        result = chain.invoke({"messages": state["messages"]})
        report = result.content if not result.tool_calls else ""
        return {"messages": [result], "market_report": report}

    return market_analyst_node
```

from __future__ import annotations

from typing import Any, Callable

from langchain_core.messages import HumanMessage
from langchain_core.prompts import ChatPromptTemplate

_SYSTEM = """You are a bullish equity researcher. Your role is to build the strongest possible
case FOR investing in the stock, drawing on all available analyst reports.

You are in a structured debate with a bearish researcher. Each round, you:
1. Review what the bear researcher said (if anything).
2. Strengthen your bullish thesis by addressing their concerns or presenting new supporting evidence.
3. Cite specific data points from the analyst reports.

Be rigorous, not cheerful. Acknowledge genuine risks but explain why the bull case prevails.
End your response with: BULL POSITION: <one-sentence summary of your stance>."""


def create_bull_researcher(llm: Any) -> Callable[[dict], dict]:
    def bull_node(state: dict) -> dict:
        ds = state["investment_debate_state"]
        context = _build_context(state, ds)
        messages = [HumanMessage(content=context)]
        result = llm.invoke(messages)
        content = result.content if hasattr(result, "content") else str(result)
